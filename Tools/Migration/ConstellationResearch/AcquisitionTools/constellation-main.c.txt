/* Development-only source-bound adapter; astronomy.c retains its original notices. */
#include "astronomy.c"
#include <stdlib.h>
#include <string.h>

static const char *mutation;
static const char *status_name(astro_status_t s)
{
    switch (s) {
        case ASTRO_SUCCESS: return "success";
        case ASTRO_INVALID_PARAMETER: return "invalid-parameter";
        case ASTRO_BAD_VECTOR: return "bad-vector";
        case ASTRO_INTERNAL_ERROR: return "internal-error";
        case ASTRO_BAD_TIME: return "bad-time";
        default: return "astronomy-error";
    }
}

static astro_deltat_func model(const char *name)
{
    if (!strcmp(name,"espenak-meeus")) return Astronomy_DeltaT_EspenakMeeus;
    if (!strcmp(name,"jpl-horizons")) return Astronomy_DeltaT_JplHorizons;
    return NULL;
}

static void print_lookup(const char *id, astro_constellation_t r)
{
    printf("{\"id\":\"%s\",\"status\":\"%s\"",id,status_name(r.status));
    if (r.status==ASTRO_SUCCESS) printf(",\"symbol\":\"%s\",\"name\":\"%s\",\"ra1875\":%.17g,\"dec1875\":%.17g",r.symbol,!strcmp(mutation,"name")?"Changed name":r.name,r.ra_1875,r.dec_1875);
    printf("}\n");
}

static int table_match(double ra,double dec)
{
    for (int i=0;i<NUM_CONSTEL_BOUNDARIES;++i) {
        int j=!strcmp(mutation,"reverse")?NUM_CONSTEL_BOUNDARIES-i-1:i;
        const constel_boundary_t *b=&ConstelBounds[j];
        int lower=!strcmp(mutation,"lower-exclusive")?b->ra_lo<ra:b->ra_lo<=ra;
        int upper=!strcmp(mutation,"upper-inclusive")?b->ra_hi>=ra:b->ra_hi>ra;
        if (b->dec_lo<=dec && lower && upper) return j;
    }
    return -1;
}

typedef struct { astro_constellation_t values[4]; } cold_result_t;
static void *cold_worker(void *pointer)
{
    cold_result_t *r=pointer;
    const double anchors[4][2]={{2.53,89.26},{5.92,7.41},{6.752,-16.716},{18.615,38.784}};
    for (int i=0;i<4;++i) r->values[i]=Astronomy_Constellation(anchors[i][0],anchors[i][1]);
    return NULL;
}

int main(int argc,char **argv)
{
    if (argc!=4 || !model(argv[1]) || !model(argv[2])) return 2;
    mutation=argv[3];Astronomy_SetDeltaTFunction(model(argv[1]));
    char line[256],id[96],op[32];double ra,dec;int count=0;
    while (fgets(line,sizeof(line),stdin)) {
        if (sscanf(line,"%95s %31s %lf %lf",id,op,&ra,&dec)!=4) return 2;
        if (!strcmp(op,"cold")) {
            pthread_t threads[32];cold_result_t results[32];
            for (int i=0;i<32;++i) if (pthread_create(&threads[i],NULL,cold_worker,&results[i])) return 3;
            for (int i=0;i<32;++i) if (pthread_join(threads[i],NULL)) return 3;
            for (int i=0;i<32;++i) for(int j=0;j<4;++j) {
                char identifier[96];snprintf(identifier,sizeof(identifier),"cold-%d-%d",i,j);print_lookup(identifier,results[i].values[j]);
            }
        } else if (!strcmp(op,"table")) {
            int row=table_match(ra,dec);
            if (row<0) printf("{\"id\":\"%s\",\"status\":\"no-table-match\"}\n",id);
            else {
                int index=ConstelBounds[row].index;
                printf("{\"id\":\"%s\",\"status\":\"success\",\"matchedRecord\":%d,\"index\":%d,\"symbol\":\"%s\",\"name\":\"%s\"}\n",id,row,index,ConstelInfo[index].symbol,!strcmp(mutation,"name")?"Changed name":ConstelInfo[index].name);
            }
        } else if (!strcmp(op,"inverse")) {
            astro_constellation_t warm=Astronomy_Constellation(2.53,89.26);
            if (warm.status!=ASTRO_SUCCESS) return 3;
            if (!strcmp(mutation,"identity-frame")) printf("{\"id\":\"%s\",\"status\":\"success\",\"raHours\":%.17g,\"decDegrees\":%.17g}\n",id,ra/360,dec/24);
            else {
                astro_spherical_t sphere={.status=ASTRO_SUCCESS,.lat=dec/24,.lon=ra/24,.dist=1};
                astro_vector_t v=Astronomy_VectorFromSphere(sphere,constel_epoch2000);
                astro_equatorial_t e=Astronomy_EquatorFromVector(Astronomy_RotateVector(Astronomy_InverseRotation(constel_rot),v));
                if(e.status!=ASTRO_SUCCESS) return 3;
                printf("{\"id\":\"%s\",\"status\":\"success\",\"raHours\":%.17g,\"decDegrees\":%.17g}\n",id,e.ra,e.dec);
            }
        } else if (!strcmp(op,"lookup")) print_lookup(id,Astronomy_Constellation(ra,dec));
        else return 2;
        if (++count==1) Astronomy_SetDeltaTFunction(model(argv[2]));
    }
    return ferror(stdin)?3:0;
}
